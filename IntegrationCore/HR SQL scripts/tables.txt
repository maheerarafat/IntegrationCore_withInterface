CREATE OR REPLACE TRIGGER update_map_record_sts
AFTER INSERT
ON tsusers
FOR EACH ROW
BEGIN
update inthrusermap set done =1 where tempuid = :new.flduid

END;

before delete
ON tsusers
FOR EACH ROW
BEGIN
delete from inthrusermap where tempuid = :new.flduid

END;
